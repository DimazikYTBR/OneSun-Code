# Язык интерфейса: en (English) или ru (Русский).
# Можно поменять здесь или командой /lang внутри программы.
LANG=ru

# Провайдер нейросети. Варианты:
#   anthropic   — Claude
#   gemini      — Google Gemini
#   openai      — OpenAI (GPT)
#   deepseek    — DeepSeek
#   openrouter  — OpenRouter (десятки моделей через один ключ)
#   custom      — любой OpenAI-совместимый API (тогда заполни BASE_URL)
PROVIDER=anthropic

# Твой API-ключ выбранного провайдера
API_KEY=

# Название модели. Примеры:
#   anthropic:  claude-sonnet-5-5 | claude-opus-5-5 | claude-haiku-5-5
#   gemini:     gemini-4-argon | gemini-4-carbon | gemini-4-hydrogen | gemini-3.8-flash
#   openai:     gpt-4.1 | gpt-4o | o4-mini
#   deepseek:   deepseek-chat | deepseek-reasoner
#   openrouter: google/gemini-4-carbon | openai/gpt-4o
MODEL=

# Нужен только для PROVIDER=custom, либо если ходишь через прокси/зеркало.
# Пример: https://my-proxy.example.com/v1

# Режим разрешений при запуске:
#   ask   — спрашивать перед правкой файлов и запуском команд (рекомендуется)
#   edits — файлы правятся без вопросов, команды спрашивают
#   yolo  — всё без вопросов (осторожно!)
MODE=ask

# Закреплённая шапка с логотипом и названием (on — включена, off — выключена).
# При включённой шапке колесо мыши не прокручивает старые сообщения за верх экрана;
# если это мешает, поставь off.
